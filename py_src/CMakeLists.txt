set(SYMUSIC_PYTHON_STUB "" CACHE FILEPATH "Pre-generated core.pyi for cross-compilation")
if(SYMUSIC_PYTHON_STUB AND NOT EXISTS "${SYMUSIC_PYTHON_STUB}")
    message(FATAL_ERROR "SYMUSIC_PYTHON_STUB must name an existing generated core.pyi")
elseif(CMAKE_CROSSCOMPILING AND NOT SYMUSIC_PYTHON_STUB)
    message(FATAL_ERROR "Cross builds require SYMUSIC_PYTHON_STUB generated by a native build")
endif()

# Linked-mode abi3 requires this optional component; older CPython and PyPy
# still build a version-specific extension.
find_package(Python REQUIRED COMPONENTS Interpreter Development.Module
    OPTIONAL_COMPONENTS Development.SABIModule)
symusic_add_dependency(nanobind)
symusic_add_dependency(pyvec)

# Only the C++ core carries LTO objects. The binding and nanobind compile
# to native objects so each Python wheel avoids optimizing that code again.
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION OFF)
set(_symusic_nb_flags NB_STATIC STABLE_ABI PROTECT_STACK NOSTRIP)
if(SYMUSIC_FAST_BUILD)
    list(APPEND _symusic_nb_flags NOMINSIZE)
endif()
# nanobind v3.1.0 headers use stdio declarations without including <cstdio>;
# libc++ does not provide them transitively, so limited-API builds against it
# fail. Directory-scoped options reach both the binding and the nanobind
# library target created below. Upstream commit 942e75d fixes this; remove
# these options with the first nanobind release containing it.
add_compile_options("$<$<COMPILE_LANGUAGE:CXX>:$<IF:$<CXX_COMPILER_ID:MSVC>,/FIcstdio,SHELL:-include cstdio>>")
nanobind_add_module(core ${_symusic_nb_flags} NB_DOMAIN symusic
    instantiations/conversion.cpp
    instantiations/midi.cpp
    instantiations/pianoroll.cpp
    instantiations/score.cpp
    instantiations/track.cpp
    instantiations/zpp.cpp
    bindings/core/module.cpp
    bindings/events/register_events.cpp
    bindings/events/register_events_notes.cpp
    bindings/events/register_events_controls.cpp
    bindings/track/register_tracks.cpp
    bindings/score/register_scores.cpp
    bindings/synth/synthesizer_bindings.cpp
)
target_link_libraries(core PRIVATE symusic pyvec)
target_include_directories(core PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${PROJECT_SOURCE_DIR}/3rdparty/zpp_bits")
target_compile_definitions(core PRIVATE $<$<BOOL:${SYMUSIC_WARN_ON_LEAKS}>:MEM_LEAK_WARNING>)
set_target_properties(core PROPERTIES
    INTERPROCEDURAL_OPTIMIZATION FALSE
    LIBRARY_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}"
    RUNTIME_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}"
)
install(TARGETS core LIBRARY DESTINATION symusic)

if(SYMUSIC_PYTHON_STUB)
    install(FILES "${SYMUSIC_PYTHON_STUB}" DESTINATION symusic RENAME core.pyi)
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/py.typed" "")
    install(FILES "${CMAKE_CURRENT_BINARY_DIR}/py.typed" DESTINATION symusic)
else()
    nanobind_add_stub(symusic_core_stub_raw INSTALL_TIME MODULE symusic.core
        OUTPUT symusic/core.raw.pyi MARKER_FILE symusic/py.typed PYTHON_PATH "."
        PATTERN_FILE "${PROJECT_SOURCE_DIR}/scripts/core_stubgen.pattern")
    configure_file("${PROJECT_SOURCE_DIR}/cmake/LiftStubs.cmake.in" LiftStubs.cmake @ONLY)
    install(SCRIPT "${CMAKE_CURRENT_BINARY_DIR}/LiftStubs.cmake")
endif()

add_subdirectory("${PROJECT_SOURCE_DIR}/cmake/abcmidi" "${PROJECT_BINARY_DIR}/abcmidi")
install(TARGETS abc2midi midi2abc RUNTIME DESTINATION symusic/_vendor/abcmidi)
install(FILES "${PROJECT_SOURCE_DIR}/3rdparty/abcmidi/doc/gpl.txt"
    DESTINATION symusic/_vendor/abcmidi RENAME LICENSE.txt)
add_subdirectory("${PROJECT_SOURCE_DIR}/cmake/fluidsynth" "${PROJECT_BINARY_DIR}/fluidsynth")

# The agent skill, located at run time by symusic.skill.path().
install(DIRECTORY "${PROJECT_SOURCE_DIR}/skills/symusic" DESTINATION symusic/skills)
